      const icon = document.createElement("div");
      icon.className = "phase-emoji";
      icon.textContent = phase.emoji;

      const body = document.createElement("div");

      const title = document.createElement("h4");
      title.textContent = `${phase.name} · ${phase.start}–${phase.end} мин`;

      const meta = document.createElement("div");
      meta.className = "meta";
      meta.textContent = `BPM ${phase.bpm[0]}–${phase.bpm[1]} · ${phase.tip}`;

      const cue = document.createElement("p");
      cue.textContent = `«${phase.cue}»`;

      body.append(title, meta, cue);
      article.append(icon, body);
      phaseResults.append(article);
    });

    emptyState.hidden = true;
    loader.hidden = true;
    loader.classList.remove("active");
    resultContent.hidden = false;
    playerStatus.classList.remove("show");
    playButton.textContent = "▶ Слушать демо";
  }

  form.addEventListener("submit", event => {
    event.preventDefault();
    error.textContent = "";

    let settings;

    try {
      settings = readSettings();
    } catch (validationError) {
      error.textContent = validationError.message;
      return;
    }

    currentScenario = null;
    const token = ++generationToken;

    generateBtn.disabled = true;
    generateBtn.textContent = "⏳ Генерируем…";

    resultContent.hidden = true;
    emptyState.hidden = true;
    loader.hidden = false;
    loader.classList.add("active");

    // Имитация генерации в течение 1,4 секунды.
    // Реальных запросов к API или ИИ нет.
    window.setTimeout(() => {
      if (token !== generationToken) return;

      currentScenario = {
        app: "AI Audio Coach — demo",
        generatedAt: new Date().toISOString(),
        mode: "mock",
        note: "Это демонстрационный сценарий. Аудио не генерируется и не воспроизводится.",
        ...settings,
        phases: buildPhases(
          settings.duration,
          settings.goal,
          settings.level,
          settings.frequency
        )
      };

      renderScenario(currentScenario);
      generateBtn.disabled = false;
      generateBtn.textContent = "✦ Сгенерировать тренировку";
    }, 1400);
  });

  // Заглушка плеера: показывает статус вместо реального звука.
  playButton.addEventListener("click", () => {
    if (!currentScenario) return;

    playerStatus.classList.toggle("show");

    playButton.textContent = playerStatus.classList.contains("show")
      ? "Ⅱ Остановить демо"
      : "▶ Слушать демо";

    if (playerStatus.classList.contains("show")) {
      playerStatus.textContent =
        "▶ Демо-плеер запущен — реальное аудио не воспроизводится.";
    }
  });

  // Экспорт сценария в JSON через Blob — полностью на стороне браузера.
  document.getElementById("downloadJson").addEventListener("click", () => {
    if (!currentScenario) return;

    const blob = new Blob(
      [JSON.stringify(currentScenario, null, 2)],
      { type: "application/json;charset=utf-8" }
    );

    const url = URL.createObjectURL(blob);
    const anchor = document.createElement("a");

    anchor.href = url;
    anchor.download = "ai-audio-coach-workout.json";

    document.body.append(anchor);
    anchor.click();
    anchor.remove();

    URL.revokeObjectURL(url);
  });
})();
